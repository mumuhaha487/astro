<h1 id="%E6%80%BB%E4%BB%A3%E7%A0%81">总代码</h1>

<p>最后附上最后的总代码（未插入数据，自己编写就行）</p>

<pre>
<code class="language-cpp">#include &lt;iostream&gt;
using namespace std;
//定义一个颜色
enum Colour
{
	RED,
	BLACK,
};
//定义一个模版T
template&lt;typename T&gt;
//一个节点的结构体
struct RBNode
{
	RBNode* lchild,
		* rchild,
		* parent;
	T	data;
	//初始化为RED也可以后面定义一个初始化函数
	Colour colour=RED;
};
//再定义一个模版T（template只能联系到下面一句语句）
template &lt;typename T&gt;
class RBTree//树
{
public:
	RBTree()
	{
		root = nullptr;
	}
	~RBTree()
	{
		ReleaseNode(root);
	}
	//插入元素
	void InsNode(const T&amp; e)
	{
		InsNode(root, e);
	}
	void InsNode(RBNode&lt;T&gt;*&amp; tnode, const T&amp; e)
	{
		RBNode&lt;T&gt;* point = tnode;
		RBNode&lt;T&gt;* parent = nullptr;
		//深入到叶子节点准备插入新数据
		while (point-&gt;data==nullptr)
		{
			if (e == point-&gt;data)
			{
				printf("该数据已经存在");
				return;
			}
			parent = point;
			if (e &gt;= point)
				point = point-&gt;rchild;
			else
				point = point-&gt;lchild;

		}
		//准备插入数据
		point = new RBNode&lt;T&gt;;
		point-&gt;data = e;
		point-&gt;lchild = nullptr;
		point-&gt;rchild = nullptr;
		point-&gt;parent = nullptr;
		point-&gt;colour = RED;
		
		if (parent == nullptr)
		{
			//插入的是根节点
			point-&gt;colour = BLACK;
			tnode = point;
		}
		//不是根节点
		if (e &gt; parent-&gt;data)
			parent-&gt;rchild = point; 
		else
			parent-&gt;lchild = point;

		point-&gt;parent = parent;
		if (parent-&gt;colour == BLACK)
		{
			count &lt;&lt; e &lt;&lt; "插入完成" &lt;&lt; endl;
			return;
		}
		//接下来就是插入地方的父亲节点是红色的，且深度至少为3（因为第二层父亲节点颜色一定为黑色
		RBNode&lt;T&gt;* uncle = nullptr;
		RBNode&lt;T&gt;* grandpa = nullptr;
		//开始整个平衡性调整
		while (true)
		{
			uncle = (parent-&gt;parent != nullptr) ? (getuncle(parent) :nullptr);
			grandpa = point-&gt;parent-&gt;parent;
			if grandpa == nullptr
				break;
			if (uncle != nullptr &amp;&amp; uncle-&gt;colour == RED)
			{
				parent-&gt;colour = BLACK;
				uncle-&gt;colour = BLACK;
				grandpa-&gt;colour = RED;
				if (grandpa == root)
				{
					grandpa-&gt;colour = BLACK;
					break;
				}
				point = grandpa;
				parent = grandpa-&gt;parent;
				if (parent-&gt;colour == BLACK)
					break;
				continue;
			}


			//（uncle != nullptr &amp;&amp; uncle-&gt;colour == RED）不满足
			int sign = 0;
			RBNode&lt;T&gt;* gff = grandpa-&gt;parent;
			//标记数据sign判断祖父节点与曾祖父节点的关系
			if (gff != nullptr)
			{
				if (gff-&gt;lchild = grandpa)
					sign = 1;
				else
					sign = 2;
			}
			//插入数据的父亲是爷爷的左节点
			if (parent-&gt;parent-&gt;lchild==parent)
			{
				//插入数据是父亲的左节点
				if (parent-&gt;lchild = point)
				{
					RotateRight(grandpa);
					grandpa-&gt;colour = BLACK;
					grandpa-&gt;rchild-&gt;colour = RED;
				}
				//插入数据是父亲的右节点
				else
				{
					RotateLeftRight(grandpa);
					grandpa-&gt;lchild = RED;
				}
			}
			else
			{
				if (parent-&gt;lchild = point)
				{
					Rotateleft(grandpa);
					grandpa-&gt;colour = BLACK;
					grandpa-&gt;lchild-&gt;colour = RED;
				}
				//插入数据是父亲的右节点
				else
				{
					RotateRightLeft(grandpa);
					grandpa-&gt;lchild = RED;
				}
			}
			if (parent-&gt;colour == BLACK)
				break;
			if (gff == nullptr)
				root = grandpa;
			else if (sign == 1)
				gff-&gt;lchild = grandpa;
			else if (sign == 2)
				gff-&gt;rchild = grandpa;

			break;
		}
		//无论如何根节点颜色为黑色
		root-&gt;colour = BLACK;
	}
private:
	//删除节点，释放内存
	void ReleaseNode(RBNode&lt;T&gt;* pnode)
	{
		if (pnode != nullptr)

	//获取叔叔节点
	RBNode&lt;T&gt;* getuncle(RBNode&lt;T&gt;* p)
	{
		if (p-&gt;parent-&gt;lchild == p)
			return p-&gt;parent-&gt;rchild;
		else

		// 3        6                     4         7
		//         /   \                 /  \
		//        5     7              3     5   
		RBNode&lt;T&gt;* node_1 = pointer;
		pointer = pointer-&gt;rchild;
		pointer-&gt;parent = node_1-&gt;parent;
		node_1-&gt;rchild = pointer-&gt;lchild;
		if (pointer-&gt;lchild)
			pointer-&gt;lchild-&gt;parent = node_1;

		RotateLeft(noderoot_left);
